### Background / Foreground Compositions

These compositions pair every `bg.*` token with plausible `text.*` and `border.*`
tokens so you can visually verify contrast. Each cell shows the approximate
[WCAG contrast ratio](https://www.w3.org/WAI/WCAG21/Understanding/contrast-minimum.html).

- **AA** (≥ 4.5 for normal text, ≥ 3 for large text) — shown in **green**
- **AA Large** (≥ 3 but < 4.5) — shown in **amber**
- **Fail** (< 3) — shown in **red**

export function hexToRgb(hex) {
  const raw = hex.replace('#', '')
  const bigint = parseInt(raw, 16)
  return [(bigint >> 16) & 255, (bigint >> 8) & 255, bigint & 255]
}

export function sRGBtoLinear(c) {
  const s = c / 255
  return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4)
}

export function luminance(hex) {
  const [r, g, b] = hexToRgb(hex)
  return 0.2126 * sRGBtoLinear(r) + 0.7152 * sRGBtoLinear(g) + 0.0722 * sRGBtoLinear(b)
}

export function contrastRatio(hex1, hex2) {
  const l1 = luminance(hex1)
  const l2 = luminance(hex2)
  const lighter = Math.max(l1, l2)
  const darker = Math.min(l1, l2)
  return (lighter + 0.05) / (darker + 0.05)
}

export function contrastLabel(ratio) {
  if (ratio >= 4.5) return 'AA'
  if (ratio >= 3) return 'AA Large'
  return 'Fail'
}

export function contrastColor(ratio) {
  if (ratio >= 4.5) return '#16a34a'
  if (ratio >= 3) return '#d97706'
  return '#dc2626'
}

### jsxstyle Integration

Pass the `mediaQueries` object to jsxstyle components. Keys become prefixes for responsive style props:

```tsx
<Block
  padding={spacing.md}
  mediaQueries={mediaQueries}
  lgPadding={spacing.xl}
/>
```

**Types:** `BreakpointToken` (key), `BreakpointValue` (value), `MediaQueryToken`, `MediaQueryValue`
